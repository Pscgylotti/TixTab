cmake_minimum_required(VERSION 3.20)
project(tixtab_ocr LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_path(TIXTAB_OCR_INCLUDE tesseract/baseapi.h REQUIRED)
find_library(TIXTAB_TESSERACT tesseract55 REQUIRED)
find_library(TIXTAB_LEPTONICA leptonica-1.85.0 REQUIRED)
add_executable(tixtab-ocr main.cpp)
target_include_directories(tixtab-ocr PRIVATE "${TIXTAB_OCR_INCLUDE}")
target_compile_definitions(tixtab-ocr PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(tixtab-ocr PRIVATE "${TIXTAB_TESSERACT}" "${TIXTAB_LEPTONICA}" windowscodecs ole32)
install(TARGETS tixtab-ocr RUNTIME DESTINATION .)
if(TIXTAB_OPENVINO_ROOT)
  add_executable(tixtab-ocr-npu main.cpp)
  target_include_directories(tixtab-ocr-npu PRIVATE "${TIXTAB_OCR_INCLUDE}" "${TIXTAB_OPENVINO_ROOT}/include")
  target_compile_definitions(tixtab-ocr-npu PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN TIXTAB_NPU)
  target_link_libraries(tixtab-ocr-npu PRIVATE "${TIXTAB_TESSERACT}" "${TIXTAB_LEPTONICA}" "${TIXTAB_OPENVINO_ROOT}/libs/openvino_c.lib" windowscodecs ole32)
endif()
